-- Add GUID column to tracks table for public-facing URLs
-- This prevents exposing the actual number of tracks in the database

-- Step 1: Add guid column (nullable initially)
ALTER TABLE tracks ADD COLUMN guid UUID;

-- Step 2: Generate UUIDs for all existing tracks
UPDATE tracks SET guid = gen_random_uuid() WHERE guid IS NULL;

-- Step 3: Make guid column NOT NULL and add unique constraint
ALTER TABLE tracks ALTER COLUMN guid SET NOT NULL;
ALTER TABLE tracks ADD CONSTRAINT tracks_guid_unique UNIQUE (guid);

-- Step 4: Add DEFAULT constraint for automatic UUID generation on INSERT
ALTER TABLE tracks ALTER COLUMN guid SET DEFAULT gen_random_uuid();

-- Step 5: Create index for fast lookups by GUID
CREATE INDEX idx_tracks_guid ON tracks(guid);

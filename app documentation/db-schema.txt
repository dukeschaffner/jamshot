

CREATE TABLE users (
  id SERIAL PRIMARY KEY,
  name VARCHAR(40) NOT NULL,
  username VARCHAR(20) UNIQUE NOT NULL,
  email VARCHAR(255) UNIQUE NOT NULL,
  password_hash VARCHAR(255) NOT NULL, -- Hashed with bcrypt
  profile_pic_url VARCHAR(255), -- S3 URL for profile pic
  bio TEXT, -- Optional short bio
  verified BOOL not null default false, -- Reserved for well-known artists
  email_verified BOOL not null default false, -- For email verification
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  is_private BOOLEAN NOT NULL DEFAULT FALSE,
  is_supporter BOOLEAN NOT NULL DEFAULT FALSE,
  terms_accepted BOOLEAN NOT NULL DEFAULT FALSE,
  privacy_policy_accepted BOOLEAN NOT NULL DEFAULT FALSE,
  policy_accepted_at TIMESTAMP,
  policy_accepted_ip VARCHAR(45),
  policy_version VARCHAR(10) DEFAULT '1.0',
  subscription_tier VARCHAR(20) NOT NULL DEFAULT 'free',
  subscription_expires_at TIMESTAMP,
  stripe_customer_id VARCHAR(255),
  stripe_subscription_id VARCHAR(255),
  date_of_birth DATE,
  analytics_consent BOOLEAN NOT NULL DEFAULT FALSE,
  analytics_consent_at TIMESTAMP,
  analytics_consent_ip VARCHAR(45),
  stripe_account_id VARCHAR(255)
);

CREATE INDEX idx_users_subscription_tier ON users(subscription_tier);
CREATE INDEX idx_users_subscription_expires_at ON users(subscription_expires_at); 



CREATE TABLE tracks (
  id SERIAL PRIMARY KEY,
  user_id INT REFERENCES users(id) ON DELETE CASCADE, -- Who uploaded it
  title VARCHAR(100) NOT NULL,
  audio_url VARCHAR(1000) NOT NULL, -- S3 URL for the clip
  combined_audio_url VARCHAR(1000),
  duration FLOAT NOT NULL, -- Seconds, for UI display
  layer INT NOT NULL, -- 0-4 (max 5 collaborators)
  parent_track_id INT REFERENCES tracks(id) ON DELETE SET NULL, -- Null for originals, links to parent for versions
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  play_count INT DEFAULT 0,
  is_private BOOLEAN NOT NULL DEFAULT FALSE,
  metronome_bpm INTEGER,
  metronome_offset FLOAT DEFAULT 0,
  time_signature VARCHAR(10) NOT NULL DEFAULT '4/4',
  secret_token VARCHAR(64), -- For secure sharing of private tracks
  allow_download BOOLEAN NOT NULL DEFAULT TRUE,
  is_competition_entry BOOLEAN NOT NULL DEFAULT FALSE,
  competition_id INT REFERENCES competitions(id) ON DELETE SET NULL -- used to indicate both competition entries and hosted competitions
  processing_status TEXT DEFAULT 'completed' CHECK (processing_status IN ('pending', 'processing', 'completed', 'failed')),
  processing_error TEXT
);

CREATE INDEX idx_tracks_processing_status ON tracks(processing_status);

-- Add function to extract mentions from text
CREATE OR REPLACE FUNCTION extract_mentions(text_content TEXT)
RETURNS TEXT[] AS $$
DECLARE
  mentions TEXT[] := '{}';
  match TEXT;
BEGIN
  -- Find all @username patterns in the text
  FOR match IN SELECT regexp_matches(text_content, '@([a-zA-Z0-9_]+)', 'g')
  LOOP
    mentions := array_append(mentions, match);
  END LOOP;
  
  RETURN mentions;
END;
$$ LANGUAGE plpgsql;

-- Functions for analytics calculations

-- Function to calculate age from date of birth
CREATE OR REPLACE FUNCTION calculate_age(birth_date DATE)
RETURNS INT AS $$
BEGIN
  RETURN EXTRACT(YEAR FROM AGE(birth_date));
END;
$$ LANGUAGE plpgsql;

-- Function to get age range from age
CREATE OR REPLACE FUNCTION get_age_range(age INT)
RETURNS VARCHAR(20) AS $$
BEGIN
  CASE 
    WHEN age < 13 THEN RETURN 'under_13';
    WHEN age BETWEEN 13 AND 17 THEN RETURN '13_17';
    WHEN age BETWEEN 18 AND 24 THEN RETURN '18_24';
    WHEN age BETWEEN 25 AND 34 THEN RETURN '25_34';
    WHEN age BETWEEN 35 AND 44 THEN RETURN '35_44';
    WHEN age BETWEEN 45 AND 54 THEN RETURN '45_54';
    WHEN age >= 55 THEN RETURN '55_plus';
    ELSE RETURN 'unknown';
  END CASE;
END;
$$ LANGUAGE plpgsql;



-- Views for common analytics queries

-- View for track performance summary
CREATE VIEW track_performance_summary AS
SELECT 
  t.id as track_id,
  t.title,
  t.user_id,
  u.username as artist_username,
  COUNT(tp.id) as total_plays,
  COUNT(DISTINCT tp.user_id) as unique_listeners,
  AVG(tp.listen_duration) as avg_listen_duration,
  COUNT(CASE WHEN tp.is_complete_play THEN 1 END) as complete_plays,
  COUNT(l.id) as total_likes,
  COUNT(c.id) as total_comments,
  COUNT(r.id) as total_reposts
FROM tracks t
LEFT JOIN users u ON t.user_id = u.id
LEFT JOIN track_plays tp ON t.id = tp.track_id
LEFT JOIN likes l ON t.id = l.track_id
LEFT JOIN comments c ON t.id = c.track_id
LEFT JOIN reposts r ON t.id = r.track_id
GROUP BY t.id, t.title, t.user_id, u.username;

-- View for user analytics summary
CREATE VIEW user_analytics_summary AS
SELECT 
  u.id as user_id,
  u.username,
  COUNT(DISTINCT t.id) as total_tracks,
  COUNT(tp.id) as total_plays_received,
  COUNT(DISTINCT tp.user_id) as total_listeners,
  AVG(tp.listen_duration) as avg_listen_duration,
  COUNT(f.follower_id) as total_followers,
  COUNT(f2.following_id) as total_following
FROM users u
LEFT JOIN tracks t ON u.id = t.user_id
LEFT JOIN track_plays tp ON t.id = tp.track_id
LEFT JOIN follows f ON u.id = f.following_id
LEFT JOIN follows f2 ON u.id = f2.follower_id
GROUP BY u.id, u.username;

-- Create competitions table
CREATE TABLE competitions (
  id SERIAL PRIMARY KEY,
  startdate TIMESTAMP NOT NULL,
  enddate TIMESTAMP NOT NULL,
  track_id INT REFERENCES tracks(id) ON DELETE CASCADE NOT NULL,
  title VARCHAR(255), -- Used for sponsored competitions only
  description TEXT, -- Used for sponsored competitions only
  prize_amount INT, -- In cents, null for sponsored competitions with non-cash prizes
  host_id INT REFERENCES users(id) ON DELETE SET NULL, -- Null for sponsored competitions
  pinned BOOLEAN NOT NULL DEFAULT FALSE,
  sponsored BOOLEAN NOT NULL DEFAULT FALSE,
  sponsor_name VARCHAR(255), -- Used for sponsored competitions only
  image_url VARCHAR(1000), -- Used for sponsored competitions only
  voucher_code VARCHAR(100), -- For gifted/preconfigured competitions
  winner_selection_method VARCHAR(20) NOT NULL CHECK (winner_selection_method IN ('curated', 'automated')),
  winner_id INT REFERENCES users(id) ON DELETE SET NULL,
  backup_winner_id INT REFERENCES users(id) ON DELETE SET NULL, -- Determined at competition end if not automated
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

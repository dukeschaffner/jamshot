40% New (last 24–48h)

30% Active (tracks with recent collabs or comments)

20% Popular (likes / plays, time-decayed)

10% Followed users




base_score = (likes*3 + collabs*5 + reposts*4 + plays*0.1)



SELECT *,
(
  base_score / POWER(EXTRACT(EPOCH FROM (NOW() - created_at))/3600 + 2, 1.3)
) AS score
FROM tracks
WHERE (
  score < :last_score
  OR (score = :last_score AND id < :last_id)
)
ORDER BY score DESC, id DESC
LIMIT 20;
